rem === Free-Motions : partitionnement UEFI/GPT du disque 0 (diskpart /s) ===
select disk 0
clean
convert gpt
create partition efi size=260
format quick fs=fat32 label="Systeme"
assign letter=S
create partition msr size=16
create partition primary
format quick fs=ntfs label="Windows"
assign letter=W
exit
